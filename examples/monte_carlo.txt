N_PATHS := 16
N_STEPS := 50
S0 := 0.0
DRIFT := 0.5
VOL := 0.4
DT := 0.1

hash(s) := fract(sin(s * 127.1 + 311.7) * 43758.5453)

sim(time_val, seed) := (
    price := S0,
    j := 0.0,
    n := floor(max(time_val, 0.0) / DT),
    while (j < n) (
        rn := (hash(seed + j * 17.31) - 0.5) * 3.46,
        price := price + DRIFT * DT + VOL * sqrt(DT) * rn,
        j := j + 1.0
    ),
    price
)

monte_carlo(x, y) := (
    bright := 0.0,
    avg := 0.0,
    i := 0.0,
    while (i < N_PATHS) (
        p := sim(x, i * 137.0),
        avg := avg + p,
        d := abs(y - p),
        bright := bright + smoothstep(0.08, 0.0, d) * 0.2,
        i := i + 1.0
    ),

    avg_p := avg / N_PATHS,
    md := abs(y - avg_p),
    mg := smoothstep(0.12, 0.0, md),

    s0_d := abs(y - S0),
    s0_line := smoothstep(0.03, 0.0, s0_d) * 0.3 * step(0.0, x),

    g := clamp(bright, 0.0, 1.0),

    (g * 0.2 + mg * 1.0 + s0_line * 0.5,
     g * 0.5 + mg * 0.7 + s0_line * 0.5,
     g * 0.9 + mg * 0.1 + s0_line * 0.5,
     1.0)
)

monte_carlo(x, y)
