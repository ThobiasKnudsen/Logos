MAX_ITER: 8,
RAD_OFFSET: 0.4,
DENOM_EPS: 0.01,
GLOW_SCALE: 6.0,
EPS: 1e-4,

warp_color(p_x, col_r, col_g, col_b, col_a): (
    (tanh(exp(p_x * 1.0) / (col_r + EPS)),
     tanh(exp(p_x * -1.0) / (col_g + EPS)),
     tanh(exp(p_x * -2.0) / (col_b + EPS)),
     tanh(exp(p_x * 0.0) / (col_a + EPS)))
),

warp(x, y): (
    dx: x.max - x.min,
    dy: y.max - y.min,
    p_x: x.min + x * dx,
    p_y: y.min + (1.0 - y) * dy,  // flips y-axis so top of screen maps to higher p_y (positive up)
    length_p: sqrt(p_x * p_x + p_y * p_y),
    denom1: length_p - RAD_OFFSET,
    denom2: abs(p_x - p_y) + DENOM_EPS,
    (v_x, v_y): (p_x / denom1 / denom2, p_y / denom1 / denom2),
    (col_r, col_g, col_b, col_a): (0.0, 0.0, 0.0, 0.0),
    i: 0.0,
    while (i < MAX_ITER) (
        i: i + 1.0,
        length_v: sqrt(v_x * v_x + v_y * v_y) + EPS,
        sin_vx: sin(v_x),
        sin_vy: sin(v_y),
        contrib: GLOW_SCALE / length_v,
        col_r: col_r + (sin_vx + 1.0) * contrib,
        col_g: col_g + (sin_vy + 1.0) * contrib,
        col_b: col_b + (sin_vy + 1.0) * contrib,
        col_a: col_a + (sin_vx + 1.0) * contrib,

        arg_y: v_x * i + time_s,
        upd_x: sin(arg_x) / i + 1.0,
        upd_y: sin(arg_y) / i + 1.0,
        v_x: v_x + upd_x,
        v_y: v_y + upd_y
    ),
    warp_color(p_x, col_r, col_g, col_b, col_a)
),
