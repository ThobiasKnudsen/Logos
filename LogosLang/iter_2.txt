create a. # create a will create a and a.type and and a.readable and a.writable. All will be set to the defensive values
a.type = struct # struct will allow for doing "create a.*"
create a.a and a.b
a.a.type = i32.
a.a.value = 0.
a.b.type = struct.
create a.b.x and a.b.y.
a.b.x.type = f64.
a.b.y.type = f64.
a.b.x.value = 2.0.
a.b.y.value = 1.0.
a.writable = true # will set all recursive structs to true as well

# if this struct setup is used many time you can create a function:
struct_init = (
	.type = fn,
	.return_type = type,
	.compiled = false,
	.updated = true,
	.semantics = (
		consumes caller{1..8},
		caller{1}.name == «(»,
		a = caller{2},
		a.type == int,
		a.size <= 32 bits,
		a.readable,
		caller{3}.name == «,»,
		b_x = caller{4},
		b_x.type == float,
		b_x.size <= 64 bits,
		b_x.readable,
		caller{5}.name == «,»,
		b_y = caller{6},
		b_y.type == float,
		b_y.size <= 64 bits,
		b_y.readable,
		caller{7}.size == «)»,
	),
	.body = (
		.type = struct,
		.a = (
			.type = int,
			.size = 32 bits,
			.offset = 0 bits,
			.value = a,
		),
		.b = (
			.type = struct,
			.x = (
				.type = float,
				.size = 64 bits,
				.offset = 0 bits,
				.value = b_x,
				.readable = true,
			),
			.y = (
				.type = float,
				.size = 64 bits,
				.offset = b_x.size,
				.value = b_y,
			),
			.offset = a.a.size,
		),
	),
),

b = struct_init(3, 2.3, 4.5),
